บล็อก #1: โค้ดสั้น 10 บรรทัด + เลขบรรทัด (Line Numbers) data-line-numbers="true" 

// Block 1: Simple function with line numbers
function calculateFactorial(n) {
    if (n <= 1) return 1;
    let result = 1;
    for (let i = 2; i <= n; i++) {
        result *= i;
    }
    return result;
}
console.log("5! =", calculateFactorial(5));

# Block 2: Python Data Pipeline with highlighted steps
import json
from datetime import datetime

def transform_record(raw_dict):
    # Step 1: Clean timestamps (HIGHLIGHTED)
    cleaned = dict(raw_dict)
    cleaned["processed_at"] = datetime.utcnow().isoformat()
    cleaned["status"] = "PROCESSED"
    
    # Step 2: Final payload (HIGHLIGHTED)
    return json.dumps(cleaned, indent=2)
