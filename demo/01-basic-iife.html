ทดสอบโหลด dist/hideko-code-block.umd.js แบบแท็ก <script> ธรรมดา (UMD/IIFE) — ตัวแปร window.HidekoCodeBlock และ window.HidekoHighlight ต้องพร้อมใช้งานทันที

JavaScript


import { useState, useEffect } from 'react';

export function App({ title = "Hideko" }) {
    const [count, setCount] = useState(0);
    useEffect(() => {
        console.log(`Count updated: ${count}`);
        document.title = `${title} - ${count}`;
    }, [count, title]);
    const handleClick = () => setCount(prev => prev + 1);

    return (
        <div className="app">
            <h1>{title}</h1>
            <button onClick={handleClick}>
                Clicked {count} times
            </button>
        </div>
    );
}
